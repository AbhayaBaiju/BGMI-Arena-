from django.shortcuts import render, redirect from django.contrib import messages from django.contrib.auth.hashers import make_password, check_password from Guest.models import ClanRegistration # ========================================================= # CHANGE CLAN PASSWORD # ========================================================= def change_password(request): # ----------------------------------------------------- # Check clan login # ----------------------------------------------------- if request.session.get("user_type") != "clan": messages.error( request, "Please login to access this page." ) return redirect("login") # ----------------------------------------------------- # Get clan ID from session # ----------------------------------------------------- clan_id = request.session.get("clan_id") if not clan_id: messages.error( request, "Clan session expired. Please login again." ) return redirect("login") # ----------------------------------------------------- # Get clan # ----------------------------------------------------- try: clan = ClanRegistration.objects.get( id=clan_id ) except ClanRegistration.DoesNotExist: request.session.flush() messages.error( request, "Clan account not found." ) return redirect("login") # ===================================================== # POST # ===================================================== if request.method == "POST": current_password = request.POST.get( "current_password", "" ).strip() new_password = request.POST.get( "new_password", "" ) confirm_password = request.POST.get( "confirm_password", "" ) # ------------------------------------------------- # Current password empty # ------------------------------------------------- if not current_password: messages.error( request, "Please enter your current password." ) return render( request, "Clan/change_password.html" ) # ------------------------------------------------- # New password empty # ------------------------------------------------- if not new_password: messages.error( request, "Please enter a new password." ) return render( request, "Clan/change_password.html" ) # ------------------------------------------------- # Confirm password empty # ------------------------------------------------- if not confirm_password: messages.error( request, "Please confirm your new password." ) return render( request, "Clan/change_password.html" ) # ------------------------------------------------- # Check current password # ------------------------------------------------- if not check_password( current_password, clan.password ): messages.error( request, "Current password is incorrect." ) return render( request, "Clan/change_password.html" ) # ------------------------------------------------- # New password length # ------------------------------------------------- if len(new_password) < 6: messages.error( request, "New password must contain at least 6 characters." ) return render( request, "Clan/change_password.html" ) # ------------------------------------------------- # Confirm password # ------------------------------------------------- if new_password != confirm_password: messages.error( request, "New password and confirm password do not match." ) return render( request, "Clan/change_password.html" ) # ------------------------------------------------- # Prevent same password # ------------------------------------------------- if check_password( new_password, clan.password ): messages.error( request, "New password must be different from your current password." ) return render( request, "Clan/change_password.html" ) # ------------------------------------------------- # Save new hashed password # ------------------------------------------------- clan.password = make_password( new_password ) clan.save( update_fields=["password"] ) # ------------------------------------------------- # Success # ------------------------------------------------- messages.success( request, "Password changed successfully." ) return redirect( "clan_profile" ) # ===================================================== # GET # ===================================================== return render( request, "Clan/change_password.html" )
